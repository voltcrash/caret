// One-time capture before retiring the unchanged synthetic legacy analysis suite.
import { afterAll, vi } from "vite-plus/test";
import { writeFileSync } from "node:fs";
import { resolve } from "node:path";

const captured = vi.hoisted(() => new Map<string, unknown>());
vi.mock("@oxytype/typing-core/events/stats", async (importOriginal) => {
  const original = await importOriginal<Record<string, unknown>>();
  return Object.fromEntries(Object.entries(original).map(([name, value]) => [
    name,
    typeof value !== "function" ? value : (...args: unknown[]) => {
      const result: unknown = value(...args);
      const log = args[0] as { context?: unknown; events?: unknown } | undefined;
      if (log?.context !== undefined && Array.isArray(log.events)) {
        const key = JSON.stringify({ name, args });
        captured.set(key, JSON.parse(JSON.stringify({ name, args, expected: result })));
      }
      return result;
    },
  ]));
});
await import("./stats.spec");
afterAll(() => {
  writeFileSync(resolve(import.meta.dirname, "../../../../packages/typing-fixtures/oracle/synthetic-analysis.json"), `${JSON.stringify([...captured.values()], null, 2)}\n`, { flag: "wx" });
});
