const synthetic: Challenge["settings"]["requirements"][] = [
  { wpm: { exact: 70 }, acc: { min: 99.5 }, raw: { exact: 70 }, con: { exact: 100 } },
  { afk: { max: 0 }, time: { min: 0 }, raw: { exact: 0 }, con: { exact: 0 } },

  {},
];
const challenges: { name: string; requirements: unknown }[] = [
  ...getChallenges().map((c) => ({ name: c.name, requirements: c.settings.requirements ?? null })),
  ...synthetic.map((requirements, i) => ({ name: `synthetic-${i}`, requirements })),
];
const cases = challenges.flatMap((challenge) =>
  Object.entries(results).flatMap(([resultName, result]) =>
    Object.entries(configs).map(([configName, config]) => ({
      challenge: challenge.name,
      result: resultName,
      config: configName,
      expected: verifyChallenge(result, config as Config, {
        settings: { requirements: challenge.requirements ?? undefined },
      } as Challenge),
    })),
  ),
);
writeFileSync(
  resolve(import.meta.dirname, "../oracle/challenges.json"),
  `${JSON.stringify({ results, configs, challenges, cases })}\n`,
  { flag: "wx" },
